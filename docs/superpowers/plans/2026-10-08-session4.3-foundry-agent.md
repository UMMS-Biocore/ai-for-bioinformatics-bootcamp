import numpy as np
import pandas as pd
import pytest

import check_tpm

DIETS, GENOS = ["chow", "hfd"], ["wt", "ALAB", "J1c", "J2c"]
SAMPLES = [f"{d}.{g}.rep{r}" for d in DIETS for g in GENOS for r in (1, 2, 3)]


def table(path, samples=SAMPLES, n=300, seed=0):
    rng = np.random.default_rng(seed)
    base = rng.lognormal(1, 2, n)
    data = {s: base * rng.lognormal(0, 0.2, n) for s in samples}
    df = pd.DataFrame({"gene": [f"g{i % (n - 20)}" for i in range(n)],
                       "transcript": [f"t{i}" for i in range(n)], **data})
    df.to_csv(path, sep="\t", index=False)
    return path


def test_good_tables_pass(tmp_path):
    out = check_tpm.run(table(tmp_path / "r.tsv"), table(tmp_path / "s.tsv", seed=1))
    assert out["own"]["rsem"]["pairs"] == 24
    assert out["shared_genes"] <= min(out["own"]["rsem"]["genes"], out["own"]["salmon"]["genes"])
    assert out["own"]["rsem"]["genes"] <= 280  # duplicate gene names were summed


def test_extra_sample_fails(tmp_path):
    with pytest.raises(AssertionError, match="24 samples"):
        check_tpm.run(table(tmp_path / "r.tsv", SAMPLES + ["chow.wt.rep4"]), table(tmp_path / "s.tsv"))


def test_group_with_two_reps_fails(tmp_path):
    bad = SAMPLES[:-1] + ["hfd.J2c.rep3x"]
    with pytest.raises(AssertionError):
        check_tpm.run(table(tmp_path / "r.tsv", bad), table(tmp_path / "s.tsv"))

`$SP/check/check_tpm.py`:

```python
"""Acceptance check for the Session 4.3 TPM tables. Prints only summary numbers.

Usage: python -I check_tpm.py RSEM.tsv SALMON.tsv OUT.json
"""
import json
import re
import sys
from itertools import combinations

import numpy as np
import pandas as pd
from scipy.stats import pearsonr, spearmanr

SAMPLE = re.compile(r"^[^.]+\.[^.]+\.rep\d+$")


def load(path):
    df = pd.read_csv(path, sep="\t")
    assert list(df.columns[:2]) == ["gene", "transcript"], "first columns must be gene, transcript"
    samples = list(df.columns[2:])
    assert len(samples) == 24, f"expected 24 samples, got {len(samples)}"
    bad = [s for s in samples if not SAMPLE.match(s)]
    assert not bad, f"sample names not <diet>.<genotype>.rep<n>: {bad}"
    df = df.drop(columns="transcript").groupby("gene").sum()
    return df.loc[(df >= 1).any(axis=1)]


def pairs(samples):
    groups = {}
    for s in samples:
        groups.setdefault(s.rsplit(".", 1)[0], []).append(s)
    assert len(groups) == 8, f"expected 8 groups, got {len(groups)}"
    out = []
    for g, reps in sorted(groups.items()):
        ps = list(combinations(sorted(reps), 2))
        assert len(ps) == 3, f"group {g}: expected 3 pairs, got {len(ps)}"
        out += [(g, a, b) for a, b in ps]
    assert len(set(out)) == 24 and all(a != b for _, a, b in out)
    return out


def corr(df, prs):
    lg = np.log2(df + 1)
    p = [pearsonr(lg[a], lg[b])[0] for _, a, b in prs]
    s = [spearmanr(df[a], df[b])[0] for _, a, b in prs]
    return p, s


def summary(df, p, s):
    r = lambda x: round(float(x), 3)
    return {"genes": len(df), "pairs": len(p),
            "pearson_median": r(np.median(p)), "pearson_min": r(min(p)),
            "spearman_median": r(np.median(s)), "spearman_min": r(min(s))}


def rsem_higher(prs, s_rsem, s_salmon):
    by = pd.DataFrame({"g": [g for g, _, _ in prs], "r": s_rsem, "s": s_salmon}).groupby("g").median()
    return int((by["r"] > by["s"]).sum())


def run(rsem_path, salmon_path):
    rsem, salmon = load(rsem_path), load(salmon_path)
    assert set(rsem.columns) == set(salmon.columns), "RSEM and Salmon must have the same samples"
    salmon = salmon[rsem.columns]
    prs = pairs(list(rsem.columns))
    shared = rsem.index.intersection(salmon.index)
    out = {"own": {}, "shared": {}, "shared_genes": len(shared)}
    cs = {}
    for tag, r, s in (("own", rsem, salmon), ("shared", rsem.loc[shared], salmon.loc[shared])):
        rp, rs = corr(r, prs)
        sp, ss = corr(s, prs)
        out[tag]["rsem"], out[tag]["salmon"] = summary(r, rp, rs), summary(s, sp, ss)
        cs[tag] = rsem_higher(prs, rs, ss)
    out["rsem_higher_spearman_groups_own"] = cs["own"]
    out["rsem_higher_spearman_groups_shared"] = cs["shared"]
    return out


if __name__ == "__main__":
    result = run(sys.argv[1], sys.argv[2])
    with open(sys.argv[3], "w") as f:
        json.dump(result, f, indent=2)
    print(json.dumps(result, indent=2))
```

- [ ] **Step 3: Write `copy.js` and its CSS**

`site/assets/copy.js`:

```js
/* Adds a Copy button to every prompt and command box on the page.
   Copies the box's own text, trimmed, as captured before the button is added.
   Where the clipboard API is unavailable, selects the text so the reader can copy it. */
(function () {
  function select(node) {
    var range = document.createRange();
    range.selectNodeContents(node);
    var sel = window.getSelection();
    sel.removeAllRanges();
    sel.addRange(range);
  }

  document.addEventListener('DOMContentLoaded', function () {
    var boxes = document.querySelectorAll('.step blockquote.prompt, .step pre.cmd');
    Array.prototype.forEach.call(boxes, function (box) {
      var text = box.textContent.trim();
      var src = document.createElement('span');
      src.className = 'copy-src';
      while (box.firstChild) src.appendChild(box.firstChild);
      box.appendChild(src);

      var btn = document.createElement('button');
      btn.type = 'button';
      btn.className = 'copy-btn';
      btn.textContent = 'Copy';
      btn.setAttribute('aria-live', 'polite');
      btn.addEventListener('click', function () {
        function done(label) {
          btn.textContent = label;
          setTimeout(function () { btn.textContent = 'Copy'; }, 1800);
        }
        function fallback() { select(src); done('Press Ctrl+C'); }
        if (navigator.clipboard && window.isSecureContext) {
          navigator.clipboard.writeText(text).then(function () { done('Copied'); }, fallback);
        } else {
          fallback();
        }
      });
      box.classList.add('has-copy');
      box.appendChild(btn);
    });
  });
})();
```
